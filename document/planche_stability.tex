The linear Stability Analysis (LST) is performed using the  inviscid Navier-stokes equation, called Euler equation, defined as  

\begin{equation} 
\frac{\partial\rho}{\partial t}+
\frac{\partial}{\partial x_i}(\rho u_i)=0
\quad \text{for} \quad i=1,2,3
\label{mass}
\end{equation} 

\begin{equation} 
\frac{\partial\rho^*}{\partial t^*}
+
u_i^*
\frac{\partial \rho^*}{\partial x_i^*} 
+
\rho^*
\frac{\partial u_i^*}{\partial x_i^*}
\quad \text{for} \quad i=1,2,3
\label{mass}
\end{equation} 

\textbf{Momemtum conservation:}

\begin{equation} 
\frac{\partial}{\partial t}(\rho u_j)
+
\frac{\partial}{\partial x_i}(\rho u_iu_j)
=
-
\frac{\partial p}{\partial x_j} 
+
\frac{\partial \tau_{ij}}{\partial x_i}
+
\rho f_j
\quad \text{for} \quad i,j=1,2,3
\end{equation} 

where 

\begin{equation} 
\tau_{ij}^*=\lambda^* \delta_{ij}^*\frac{\partial u_k^*}{\partial x_k^*}+
\mu^*\left(\frac{\partial u_i^*}{\partial x_j^*}+\frac{\partial u_j^*}{\partial x_i^*}\right)
\end{equation} 

\begin{equation*} 
u_j^*
\frac{\partial}{\partial t^*}\rho^*
+
\rho^*
\frac{\partial}{\partial t^*}u_j^*
+
\rho^* u_j^*
\frac{\partial}{\partial x_i^*}u_i^*
+
u_i^* u_j^*
\frac{\partial}{\partial x_i^*}\rho^*
+
u_i^* \rho^*
\frac{\partial}{\partial x_i} u_j^*
=
-
\frac{\partial p^*}{\partial x_j^*} 
\end{equation*} 

\begin{equation*} 
u_j^*
\cancelto{0}
{
\left(
    \frac{\partial}{\partial t^*}\rho^*
    +
    \rho^* 
    \frac{\partial}{\partial x_i}u_i^*
    +
    u_i^*
    \frac{\partial}{\partial x_i}\rho^* 
\right)
}
+
\rho^*
\frac{\partial^*}{\partial t^*}u_j^*
+
u_i^* \rho^*
\frac{\partial}{\partial x_i}u_j^*
=
-
\frac{\partial p^*}{\partial x_j^*} 
\end{equation*} 

\begin{equation} 
\rho^*
\left(
\frac{\partial}{\partial t^*}u_j^*
+
u_i^* 
\frac{\partial}{\partial x_i}u_j^*
\right)
=
-
\frac{\partial p^*}{\partial x_j^*} 
\end{equation} 

\textbf{Species conservation:}
\begin{equation} 
\rho^*
\left(
\frac{\partial}{\partial t^*}Z^*
+
u_i ^*
\frac{\partial}{\partial x_i^*}Z^*
\right)
=
0
\end{equation} 


\textbf{Energy conservation:}

\begin{equation}
\rho^*
\frac{D  h^* }{Dt^*}
=
\frac{D p^* }{Dt^*}
+
\frac{\partial u_j^*}{\partial x^*_i}
\tau_{ij}^*
+
\frac{\partial}{\partial x_i^*}
\left(
    k^*\frac{\partial T^*}{x_i^*}
\right)
-
\frac{\partial}{\partial x_i^*}
\left(
        \sum_i
        h_{si}^*
        \rho^*
        D_i^*
        \frac{\partial Y_i^*}{\partial x_i^*}
\right)
\end{equation} 

with  $\dfrac{D}{D t^*}=\dfrac{\partial}{\partial t^*}+u_i^*\dfrac{\partial }{\partial x^*_i}$, 

\begin{equation*}
\rho
\frac{\partial h}{\partial t} 
+
\rho^* u^* \frac{\partial h^*}{\partial x^*}+
\rho^* v^* \frac{\partial h^*}{\partial y^*}
=
\frac{\partial p^*}{\partial t^*}
+
u_i^*
\frac{\partial p^*}{\partial x^*_i}
+
\frac{\partial u_j^*}{\partial x^*_i}
\tau_{ij}^*
+
\frac{\partial}{\partial x_i^*}
\left(
    k^*\frac{\partial T^*}{x_i^*}
\right)
-
\frac{\partial}{\partial x_i^*}
\left(
        \sum_i
        h_{si}^*
        \rho^*
        D_i^*
        \frac{\partial Y_i^*}{\partial x_i^*}
\right)
\end{equation*} 

where $h_{si}^*$ is the sensible enthalpy 
for the specie $i$ in a multi-component misture.  

For multiples species $h_s^*=\sum Y_i^*h_{si}^*(T^*)$ with  

\begin{equation} 
h_{si}^*
=
\int_{T_0}^{T}
c_{pi}^*dT^*
\end{equation} 

and the formation enthalpy of the misture  
can be defined 

\begin{equation} 
   h_{f}^\circ

   \sum
   h_{fi}^{\circ*}
   .
\end{equation} 

Enthalpy can be split explicitly in two parts 

\begin{equation} 
    h^*=h_s^*+h_f^{\circ*}
\end{equation} 

for multiple species 

\begin{equation}
