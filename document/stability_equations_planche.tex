
To defined the linear stability equation is assuming that 
the flow can be descomposed  into  a sum of mean flow and 
small perturbation of order $\epsilon$, where the base flow 
is only function of the vertical coordinate $y$,  

\begin{equation}
\begin{split}
   \rho(t,x,y,z)=\overline{\rho}(y)+\epsilon\rho^\prime(t,x,y,z) 
   ,
   \quad
   p(t,x,y,z)=1+\epsilon p^\prime(t,x,y,z) 
   ,
   \\
   T(t,x,y,z)=\overline{T}(y)+\epsilon T^\prime(t,x,y,z) 
   ,
   \quad
   Z(t,x,y,z)=\overline{Z}(y)+\epsilon Z^\prime(t,x,y,z) 
    \\
   \quad
   R(t,x,y,z)=\overline{R}(Z)+\epsilon R^\prime(t,x,y,z) 
   ,
   u(t,x,y,z)=\overline{u}(y)+\epsilon u^\prime(t,x,y,z) 
   ,
   \quad
   \\
   v(t,x,y,z)=\epsilon v^\prime(t,x,y,z) 
   ,
   \quad
   w(t,x,y,z)=\epsilon w^\prime(t,x,y,z) 
   .
\end{split}
\label{perturbation}
\end{equation}
